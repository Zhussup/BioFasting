# Benchmark environment (PLAN.md step 0.3).
#
#   python3 -m venv .venv
#   .venv/bin/pip install -r bench/requirements.txt
#   .venv/bin/python bench/run.py
#
# Pinned to the versions the published numbers were measured with.  pysam and
# pyfaidx are optional: bench/run.py degrades to the pure-Python and Biopython
# comparisons, and says which backends were unavailable, when they are missing.
#
# biopython is a hard requirement -- it is the reference implementation the
# verification digests are compared against.
biopython==1.88
pysam==0.24.1
pyfaidx==0.9.0.4

# Step 0.4.  cProfile is in the standard library; py-spy adds an
# uninstrumented, wall-clock view that sees inside C extensions.  It is a
# profiler, not a benchmark dependency -- bench/run.py does not need it.
py-spy==0.4.2

# Charts.  Also not needed to measure anything -- the README's images are
# rendered from the runner's and profiler's own numbers by
# plot_g1.py / plot_profile.py, and the tables stay the data source.
matplotlib==3.11.2

# The Arrow tables (PLAN 2.3).  Bench/rank.py and bench/rank_arrow.py are
# differential: pyarrow holds the tables both scripts compare value by value,
# polars is the hand-off `pl.from_arrow` is measured through, and polars-bio is
# the competitor whose columns this package reproduces and whose 0.970 us a read
# was the target.  None of the three is a dependency of biofasting itself --
# pyarrow is an extra (`pip install biofasting[arrow]`) and polars is imported by
# the benchmark only -- so a machine that only runs the older runners can leave
# these out and lose those two scripts and nothing else.
pyarrow==24.0.0
polars==1.44.2
polars-bio==0.36.0
