# Build and test environment for the C++ core (step 1.1).
#
#     python3 -m venv .venv
#     source .venv/bin/activate          # needed: see the note below
#     pip install -r requirements-dev.txt
#     pip install --no-build-isolation -e .
#     pytest
#
# Activation is not cosmetic.  With editable.rebuild = true, importing the
# package runs `cmake --build` on the C++ sources, and it looks for cmake and
# ninja on PATH -- `.venv/bin/pytest` alone does not put them there, and the
# failure is a bare FileNotFoundError from inside scikit-build-core's import
# hook.  --no-build-isolation matters too: an isolated build configures the
# build directory with a cmake in a temporary environment that is deleted
# straight afterwards, so the next import tries to re-run a binary that no
# longer exists.
#
# The same .venv is used by bench/requirements.txt; installing build tools into
# it does not move a benchmark number, and one environment is one thing to
# remember.  See bench/README.md for the corpus.
#
# These are pinned to the versions the scaffold was actually built and tested
# with.  pyproject.toml carries the ranges instead, because a wheel should build
# on whatever is current rather than on one developer's snapshot -- the two
# deliberately differ, and CI builds with the ranges to prove they still work.
#
# nanobind 3.x is what sets the Python floor at 3.10 (3.1.0 declares
# Requires-Python >=3.10); that floor is declared in pyproject.toml as well.
nanobind==3.1.0
scikit-build-core==1.1.1
cmake==4.4.3
ninja==1.13.2
pytest==9.1.1
# Release tooling, here so that the cibuildwheel configuration in pyproject.toml
# can be validated locally without Docker:
#     cibuildwheel --platform linux --print-build-identifiers
build==1.6.1
cibuildwheel==4.2.1
